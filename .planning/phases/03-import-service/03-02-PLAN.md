```typescript
import { Transform, TransformCallback } from 'node:stream'
import type { Variant } from '../../database/types'
import type { DatabaseService } from '../../database/DatabaseService'
import type { ProgressCallback } from '../types'

type MappedVariant = Omit<Variant, 'id' | 'case_id'>

interface BatchAccumulatorOptions {
  caseId: number
  batchSize: number
  db: DatabaseService
  onProgress?: ProgressCallback
  startTime: number
}

export class BatchAccumulator extends Transform {
  private batch: MappedVariant[] = []
  private totalInserted = 0
  private skipped = 0
  private readonly caseId: number
  private readonly batchSize: number
  private readonly db: DatabaseService
  private readonly onProgress?: ProgressCallback
  private readonly startTime: number

  constructor(options: BatchAccumulatorOptions) {
    super({ objectMode: true })
    this.caseId = options.caseId
    this.batchSize = options.batchSize
    this.db = options.db
    this.onProgress = options.onProgress
    this.startTime = options.startTime
  }

  _transform(
    chunk: MappedVariant | null,
    encoding: BufferEncoding,
    callback: TransformCallback
  ): void {
    // Null chunks indicate skipped variants from FieldMapper
    if (chunk === null) {
      this.skipped++
      callback()
      return
    }

    this.batch.push(chunk)

    if (this.batch.length >= this.batchSize) {
      this.flushBatch()
    }

    callback()
  }

  _flush(callback: TransformCallback): void {
    // Insert any remaining variants
    if (this.batch.length > 0) {
      this.flushBatch()
    }
    callback()
  }

  private flushBatch(): void {
    if (this.batch.length === 0) return

    this.db.insertVariantsBatch(this.caseId, this.batch)
    this.totalInserted += this.batch.length

    if (this.onProgress) {
      this.onProgress({
        phase: 'inserting',
        count: this.totalInserted,
        elapsed: Date.now() - this.startTime,
        skipped: this.skipped
      })
    }

    this.batch = []
  }

  get inserted(): number {
    return this.totalInserted
  }

  get skippedCount(): number {
    return this.skipped
  }
}

export function createBatchAccumulator(options: BatchAccumulatorOptions): BatchAccumulator {
  return new BatchAccumulator(options)
}
```

```typescript
import { createReadStream, statSync } from 'node:fs'
import { pipeline } from 'node:stream/promises'
import { createGunzip } from 'node:zlib'
import { parser } from 'stream-json'
import { streamArray } from 'stream-json/streamers/StreamArray'
import { pick } from 'stream-json/filters/Pick'

import type { DatabaseService } from '../database/DatabaseService'
import { DatabaseError } from '../database/errors'
import type { ImportOptions, ImportResult, DataDictionaries } from './types'
import { createFieldMapper } from './transforms/FieldMapper'
import { createBatchAccumulator, BatchAccumulator } from './transforms/BatchAccumulator'

const DEFAULT_BATCH_SIZE = 5000

export class ImportService {
  constructor(private readonly db: DatabaseService) {}

  async importVariants(filePath: string, options: ImportOptions): Promise<ImportResult> {
    const startTime = Date.now()
    const errors: string[] = []

    // Get file size for case record
    let fileSize: number
    try {
      fileSize = statSync(filePath).size
    } catch (error) {
      throw new DatabaseError(
        `Failed to read file: ${filePath}`,
        error instanceof Error ? error : undefined
      )
    }

    // Create case record first
    const caseId = this.db.createCase(options.caseName, filePath, fileSize)

    let batchAccumulator: BatchAccumulator | null = null

    try {
      // Report initial progress
      if (options.onProgress) {
        options.onProgress({
          phase: 'reading',
          count: 0,
          elapsed: 0
        })
      }

      // Parse header to extract data dictionaries
      const dictionaries = await this.extractDictionaries(filePath)

      if (options.onProgress) {
        options.onProgress({
          phase: 'parsing',
          count: 0,
          elapsed: Date.now() - startTime
        })
      }

      // Create pipeline components
      const fieldMapper = createFieldMapper(dictionaries)
      batchAccumulator = createBatchAccumulator({
        caseId,
        batchSize: options.batchSize ?? DEFAULT_BATCH_SIZE,
        db: this.db,
        onProgress: options.onProgress,
        startTime
      })

      // Build and execute pipeline
      // The JSON format is: { "caseId": { "header": [...], "data": [[...], ...] } }
      // We need to pick the data array from inside the first (only) key
      await pipeline(
        createReadStream(filePath),
        createGunzip(),
        parser(),
        pick({ filter: /^\w+\.data/ }),  // Match "1814.data" or any key.data
        streamArray(),
        fieldMapper,
        batchAccumulator,
        { signal: options.signal }
      )

      // Update case with final variant count
      this.db.updateCaseVariantCount(caseId, batchAccumulator.inserted)

      return {
        caseId,
        variantCount: batchAccumulator.inserted,
        skipped: batchAccumulator.skippedCount,
        errors,
        elapsed: Date.now() - startTime
      }
    } catch (error) {
      // Rollback: delete the case on failure
      try {
        this.db.deleteCase(caseId)
      } catch {
        // Ignore cleanup errors
      }

      if (error instanceof Error && error.name === 'AbortError') {
        errors.push('Import cancelled by user')
        return {
          caseId: -1,
          variantCount: batchAccumulator?.inserted ?? 0,
          skipped: batchAccumulator?.skippedCount ?? 0,
          errors,
          elapsed: Date.now() - startTime
        }
      }

      throw new DatabaseError(
        `Import failed: ${error instanceof Error ? error.message : String(error)}`,
        error instanceof Error ? error : undefined
      )
    }
  }

  /**
   * Extract data dictionaries from the JSON header
   * The header contains dataDictionary fields for Gene and Impact columns
   */
  private async extractDictionaries(filePath: string): Promise<DataDictionaries> {
    const dictionaries: DataDictionaries = {
      gene: {},
      impact: {}
    }

    return new Promise((resolve, reject) => {
      const stream = createReadStream(filePath)
        .pipe(createGunzip())
        .pipe(parser())

      let headerFound = false
      let inHeader = false
      let depth = 0
      let currentColumn: { id?: string; dataDictionary?: Record<string, string> } = {}

      stream.on('data', (data: { name: string; value?: unknown }) => {
        // Track when we're inside header array
        if (data.name === 'keyValue' && data.value === 'header') {
          inHeader = true
          depth = 0
          return
        }

        if (inHeader) {
          if (data.name === 'startArray' && depth === 0) {
            depth = 1
            return
          }
          if (data.name === 'startObject' && depth === 1) {
            currentColumn = {}
            return
          }
          if (data.name === 'keyValue' && data.value === 'id') {
            // Next stringValue will be the column id
          }
          if (data.name === 'stringValue' && !currentColumn.id) {
            currentColumn.id = data.value as string
          }
          if (data.name === 'keyValue' && data.value === 'dataDictionary') {
            // Next object will be the dictionary
          }
          if (data.name === 'endObject' && depth === 1) {
            // End of column definition
            if (currentColumn.id === 'Gene' && currentColumn.dataDictionary) {
              dictionaries.gene = currentColumn.dataDictionary
            }
            currentColumn = {}
          }
          if (data.name === 'endArray' && depth === 1) {
            // End of header array
            headerFound = true
            stream.destroy()
            resolve(dictionaries)
          }
        }
      })

      stream.on('error', reject)
      stream.on('end', () => {
        if (!headerFound) {
          // Use empty dictionaries if header not found (object-per-variant format)
          resolve(dictionaries)
        }
      })
    })
  }
}
```

**Revised approach for extractDictionaries:**

```typescript
private async extractDictionaries(filePath: string): Promise<DataDictionaries> {
  // Read just enough of the file to get the header
  // The header is a small JSON array of column definitions
  const { pick } = await import('stream-json/filters/Pick')
  const { streamValues } = await import('stream-json/streamers/StreamValues')

  const dictionaries: DataDictionaries = {
    gene: {},
    impact: {}
  }

  return new Promise((resolve, reject) => {
    let resolved = false

    const stream = createReadStream(filePath)
      .pipe(createGunzip())
      .pipe(parser())
      .pipe(pick({ filter: /^\w+\.header$/ }))  // Match "1814.header"
      .pipe(streamValues())

    stream.on('data', (data: { value: Array<{ id: string; dataDictionary?: Record<string, string> }> }) => {
      // data.value is the header array
      const header = data.value
      for (const col of header) {
        if (col.id === 'Gene' && col.dataDictionary) {
          dictionaries.gene = col.dataDictionary
        }
        // Impact dictionary is static (already defined), but extract if present
        if (col.id === 'Impact' && col.dataDictionary) {
          dictionaries.impact = col.dataDictionary
        }
      }
      resolved = true
      stream.destroy()
      resolve(dictionaries)
    })

    stream.on('error', (err) => {
      if (!resolved) reject(err)
    })

    stream.on('end', () => {
      if (!resolved) {
        // No header found - assume object-per-variant format
        resolve(dictionaries)
      }
    })
  })
}
```

```typescript
import { describe, it, expect, beforeEach, afterEach, vi } from 'vitest'
import { join } from 'node:path'
import { DatabaseService } from '../../../src/main/database/DatabaseService'
import { ImportService } from '../../../src/main/import/ImportService'
import type { ProgressUpdate } from '../../../src/main/import/types'

// Test data paths
const SAMPLE_FILE = join(__dirname, '../../../../test-data/case-892-snv-sample.json.gz')
const FULL_FILE = join(__dirname, '../../../../test-data/case-892-snv-annotations.json.gz')

describe('ImportService', () => {
  let db: DatabaseService
  let importService: ImportService

  beforeEach(() => {
    db = new DatabaseService(':memory:')
    importService = new ImportService(db)
  })

  afterEach(() => {
    db.close()
  })

  describe('importVariants', () => {
    it('should import sample file and create case', async () => {
      const result = await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Test Case'
      })

      expect(result.caseId).toBeGreaterThan(0)
      expect(result.variantCount).toBe(251) // Sample has 251 variants
      expect(result.skipped).toBe(0)
      expect(result.errors).toHaveLength(0)

      // Verify case was created
      const caseRecord = db.getCase(result.caseId)
      expect(caseRecord.name).toBe('Test Case')
      expect(caseRecord.variant_count).toBe(251)
    })

    it('should extract Gene symbols from data dictionary', async () => {
      const result = await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Gene Test'
      })

      // Query variants and check gene symbols are resolved
      const variants = db.getVariants({ case_id: result.caseId }, 10)
      expect(variants.data.length).toBeGreaterThan(0)

      // Gene symbols should be strings like 'LOC100132287', not numeric IDs
      const firstVariant = variants.data[0]
      if (firstVariant.gene_symbol) {
        expect(typeof firstVariant.gene_symbol).toBe('string')
        expect(firstVariant.gene_symbol).not.toMatch(/^\d+$/) // Not just numbers
      }
    })

    it('should resolve Impact codes to labels', async () => {
      const result = await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Impact Test'
      })

      const variants = db.getVariants({ case_id: result.caseId }, 100)

      // All consequences should be resolved labels
      const consequences = variants.data
        .map(v => v.consequence)
        .filter(Boolean) as string[]

      const validLabels = ['HIGH', 'MODERATE', 'LOW', 'MODIFIER']
      for (const cons of consequences) {
        expect(validLabels).toContain(cons)
      }
    })

    it('should report progress during import', async () => {
      const progressUpdates: ProgressUpdate[] = []

      await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Progress Test',
        onProgress: (update) => progressUpdates.push({ ...update })
      })

      // Should have multiple progress updates
      expect(progressUpdates.length).toBeGreaterThan(0)

      // Should have reading phase
      expect(progressUpdates.some(p => p.phase === 'reading')).toBe(true)

      // Should have inserting phase with counts
      const insertingUpdates = progressUpdates.filter(p => p.phase === 'inserting')
      expect(insertingUpdates.length).toBeGreaterThan(0)

      // Count should increase
      if (insertingUpdates.length > 1) {
        expect(insertingUpdates[1].count).toBeGreaterThan(insertingUpdates[0].count)
      }
    })

    it('should handle cancellation with AbortSignal', async () => {
      const controller = new AbortController()

      // Abort after a short delay
      setTimeout(() => controller.abort(), 10)

      const result = await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Abort Test',
        signal: controller.signal
      })

      // Should have cancelled
      expect(result.errors).toContain('Import cancelled by user')
      expect(result.caseId).toBe(-1)
    })

    it('should rollback case on critical failure', async () => {
      // Try to import non-existent file
      await expect(
        importService.importVariants('/nonexistent/file.json.gz', {
          caseName: 'Fail Test'
        })
      ).rejects.toThrow()

      // Case should not exist
      expect(db.getAllCases()).toHaveLength(0)
    })

    it('should use custom batch size', async () => {
      const progressUpdates: ProgressUpdate[] = []

      await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Batch Test',
        batchSize: 50, // Small batch to see more progress updates
        onProgress: (update) => progressUpdates.push({ ...update })
      })

      // With 251 variants and batch size 50, should have ~5 insert updates
      const insertUpdates = progressUpdates.filter(p => p.phase === 'inserting')
      expect(insertUpdates.length).toBeGreaterThanOrEqual(4)
    })
  })

  describe('performance', () => {
    it('should import 65k variants in under 30 seconds', async () => {
      const startTime = Date.now()

      const result = await importService.importVariants(FULL_FILE, {
        caseName: 'Performance Test'
      })

      const elapsed = Date.now() - startTime

      // Full file has ~65k variants
      expect(result.variantCount).toBeGreaterThan(60000)
      expect(elapsed).toBeLessThan(30000) // Under 30 seconds
      expect(result.skipped).toBe(0)

      console.log(`Imported ${result.variantCount} variants in ${elapsed}ms`)
    }, 60000) // 60 second timeout for this test
  })

  describe('error handling', () => {
    it('should handle duplicate case names', async () => {
      // First import
      await importService.importVariants(SAMPLE_FILE, {
        caseName: 'Duplicate'
      })

      // Second import with same name should fail
      await expect(
        importService.importVariants(SAMPLE_FILE, {
          caseName: 'Duplicate'
        })
      ).rejects.toThrow(/UNIQUE constraint/)
    })
  })
})
```
