```typescript
/**
 * Composable for case metadata state management
 *
 * Provides reactive metadata state per case with IPC-backed persistence.
 * Used by CaseList, CaseMetadataCard for status, cohort, and HPO display/editing.
 */

import { ref } from 'vue'
import type {
  CaseMetadata,
  CohortGroup,
  CaseHpoTerm,
  AffectedStatus,
  FullCaseMetadata
} from '../../../shared/types/api'
```

**State management:**
- `metadataCache = ref<Map<number, FullCaseMetadata>>(new Map())` - Cache by caseId
- `loadingStates = ref<Map<number, boolean>>(new Map())` - Loading state per case
- `cohortGroupsCache = ref<CohortGroup[]>([])` - Global cohort groups list

**Core methods:**

`loadMetadata(caseId: number): Promise<void>`
- Skip if already cached or loading
- Call window.api.caseMetadata.getFullMetadata(caseId)
- Store in cache

`loadCohortGroups(): Promise<void>`
- Call window.api.caseMetadata.listCohorts()
- Store in cohortGroupsCache

`getMetadata(caseId: number): FullCaseMetadata | undefined`
- Return from cache

`isLoading(caseId: number): boolean`
- Return loading state

**Status methods:**

`updateStatus(caseId: number, status: AffectedStatus): Promise<void>`
- Optimistic update: update cache immediately
- Call window.api.caseMetadata.upsert(caseId, { affected_status: status })
- On error: revert cache

**Cohort methods:**

`setCaseCohorts(caseId: number, cohortIds: number[]): Promise<void>`
- Optimistic update: filter cohortGroupsCache by ids, update cache
- Call window.api.caseMetadata.setCohorts(caseId, cohortIds)
- On error: reload full metadata

`createAndAssignCohort(caseId: number, name: string): Promise<CohortGroup>`
- Call window.api.caseMetadata.createCohort(name)
- Add to cohortGroupsCache
- Call window.api.caseMetadata.assignCohort(caseId, newCohort.id)
- Update cache with new cohort
- Return new cohort

`getOrCreateCohort(name: string): Promise<CohortGroup>`
- Check if exists in cohortGroupsCache by name
- If exists, return it
- Otherwise call window.api.caseMetadata.createCohort(name)
- Add to cohortGroupsCache
- Return cohort

**HPO methods:**

`assignHpoTerm(caseId: number, hpoId: string, hpoLabel: string): Promise<void>`
- Optimistic update: add to cache hpoTerms array
- Call window.api.caseMetadata.assignHpoTerm
- On error: remove from cache

`removeHpoTerm(caseId: number, hpoId: string): Promise<void>`
- Optimistic update: remove from cache hpoTerms array
- Call window.api.caseMetadata.removeHpoTerm
- On error: restore to cache

**Utility methods:**

`clearCache(): void` - Clear all caches (call on database switch)
`invalidateCase(caseId: number): void` - Remove single case from cache (force reload)

Cohort color function (deterministic hash-based):
```typescript
export function getCohortColor(name: string): string {
  const colors = ['primary', 'secondary', 'success', 'info', 'warning', 'purple', 'pink', 'indigo', 'teal', 'cyan']
  let hash = 0
  for (let i = 0; i < name.length; i++) {
    hash = name.charCodeAt(i) + ((hash << 5) - hash)
  }
  return colors[Math.abs(hash) % colors.length]
}
```

Script setup:
```typescript
import { getCohortColor } from '../composables/useCaseMetadata'
import type { CohortGroup } from '../../../shared/types/api'

const props = defineProps<{
  modelValue: CohortGroup[]
  availableCohorts: CohortGroup[]
  disabled?: boolean
}>()

const emit = defineEmits<{
  'update:modelValue': [cohorts: CohortGroup[]]
  'create:cohort': [name: string]
}>()

function handleSelectionChange(newValue: (CohortGroup | string)[]) {
  // Separate existing cohorts from new names (strings)
  const existingCohorts: CohortGroup[] = []
  const newNames: string[] = []

  for (const item of newValue) {
    if (typeof item === 'string') {
      // User typed a new name
      newNames.push(item.trim())
    } else {
      existingCohorts.push(item)
    }
  }

  // Emit create event for each new name
  for (const name of newNames) {
    if (name.length > 0) {
      emit('create:cohort', name)
    }
  }

  // Emit update with existing cohorts (parent handles adding newly created ones)
  emit('update:modelValue', existingCohorts)
}
```

Note: The parent component (CaseMetadataCard) will handle the create:cohort event by calling createAndAssignCohort from the composable, then the newly created cohort will be added to the selection.
