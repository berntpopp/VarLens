    **2. Create TempDirectoryManager** (`src/main/import/TempDirectoryManager.ts`):

    A utility class managing temp directory lifecycle:

    ```typescript
    import { mkdtempSync, rmSync } from 'node:fs'
    import { join } from 'node:path'
    import { tmpdir } from 'node:os'

    export class TempDirectoryManager {
      private tempDir: string | null = null

      /**
       * Create a unique temporary directory.
       * Uses os.tmpdir() + 'varlens-zip-' prefix.
       * @returns Absolute path to created temp directory
       */
      create(): string {
        // IMPORTANT: Use path.join to ensure correct separator (Pitfall 2 from RESEARCH.md)
        this.tempDir = mkdtempSync(join(tmpdir(), 'varlens-zip-'))
        return this.tempDir
      }

      /**
       * Clean up the temporary directory.
       * Safe to call multiple times. Uses force option to suppress errors.
       */
      cleanup(): void {
        if (this.tempDir !== null) {
          try {
            rmSync(this.tempDir, { recursive: true, force: true })
          } catch (error) {
            console.error('Failed to clean up temp directory:', error)
          }
          this.tempDir = null
        }
      }

      /** Get the current temp directory path, or null if not created */
      getPath(): string | null {
        return this.tempDir
      }
    }
    ```

    **3. Create ZipExtractor** (`src/main/import/ZipExtractor.ts`):

    ```typescript
    import AdmZip from 'adm-zip'
    import { resolve, relative } from 'node:path'

    export interface ZipExtractionResult {
      extractedFiles: string[]  // Full paths to extracted JSON.gz files
      errors: string[]          // Any extraction errors
      totalEntries: number      // Total entries in archive
    }

    export class ZipExtractor {
      /**
       * Check if a ZIP file is password-protected.
       * Uses the encrypted flag in entry headers.
       *
       * @param zipPath - Path to ZIP file
       * @returns true if any entry is encrypted
       */
      isEncrypted(zipPath: string): boolean {
        try {
          const zip = new AdmZip(zipPath)
          const entries = zip.getEntries()
          return entries.some(entry => entry.header.encrypted)
        } catch {
          // If we can't read the ZIP at all, assume we need to try with password
          return false
        }
      }

      /**
       * Extract JSON.gz files from a ZIP archive to a target directory.
       *
       * Validates all entry paths to prevent Zip Slip attacks:
       * - Rejects paths containing '..'
       * - Rejects absolute paths (starting with / or \)
       * - Rejects UNC paths (Windows drive letters like C:)
       * - Verifies resolved path stays within target directory
       *
       * @param zipPath - Path to the ZIP file
       * @param targetDir - Directory to extract files into (must already exist)
       * @param password - Optional password for encrypted archives
       * @returns Extraction result with file paths and any errors
       * @throws Error if ZIP file cannot be opened or password is incorrect
       */
      extract(zipPath: string, targetDir: string, password?: string): ZipExtractionResult {
        const zip = new AdmZip(zipPath)
        const entries = zip.getEntries()
        const result: ZipExtractionResult = {
          extractedFiles: [],
          errors: [],
          totalEntries: entries.length
        }

        for (const entry of entries) {
          // Skip directories
          if (entry.isDirectory) continue

          const entryName = entry.entryName

          // Skip non-JSON.gz files
          const lowerName = entryName.toLowerCase()
          if (!lowerName.endsWith('.json.gz') && !lowerName.endsWith('.gz') && !lowerName.endsWith('.json')) {
            continue
          }

          // Zip Slip prevention
          if (!this.validatePath(targetDir, entryName)) {
            result.errors.push(`Rejected path traversal attempt: ${entryName}`)
            continue
          }

          try {
            // Extract to target directory
            // extractEntryTo params: entry, targetPath, maintainEntryPath, overwrite, keepOriginalPermission, password
            zip.extractEntryTo(entry, targetDir, false, true, false, password ?? undefined)

            // Get the basename for the extracted file path
            const basename = entryName.split('/').pop() ?? entryName
            const extractedPath = resolve(targetDir, basename)
            result.extractedFiles.push(extractedPath)
          } catch (error) {
            const errorMsg = error instanceof Error ? error.message : String(error)
            result.errors.push(`Failed to extract ${entryName}: ${errorMsg}`)
          }
        }

        return result
      }

      /**
       * Validate an entry path to prevent Zip Slip path traversal.
       * Defense-in-depth: checks multiple attack vectors.
       */
      private validatePath(targetDir: string, entryPath: string): boolean {
        // Reject path traversal sequences
        if (entryPath.includes('..')) return false

        // Reject absolute paths (Unix)
        if (entryPath.startsWith('/')) return false

        // Reject absolute paths (Windows backslash)
        if (entryPath.startsWith('\\')) return false

        // Reject UNC paths and drive letters (Windows)
        if (/^[a-zA-Z]:/.test(entryPath)) return false

        // Final check: resolved path must be within target directory
        const normalizedTarget = resolve(targetDir)
        const resolvedEntry = resolve(normalizedTarget, entryPath)
        const rel = relative(normalizedTarget, resolvedEntry)

        // relative() returns '..' prefix if path escapes target
        if (rel.startsWith('..')) return false

        return true
      }

      /**
       * Test if a ZIP archive can be opened with the given password.
       * Attempts to read the first entry as a verification check.
       *
       * @param zipPath - Path to ZIP file
       * @param password - Password to test
       * @returns true if password is correct (or archive is not encrypted)
       */
      testPassword(zipPath: string, password: string): boolean {
        try {
          const zip = new AdmZip(zipPath)
          const entries = zip.getEntries()
          if (entries.length === 0) return true

          // Try to decompress first non-directory entry with password
          const firstFile = entries.find(e => !e.isDirectory)
          if (firstFile === undefined) return true

          // getData with password - throws if wrong password
          firstFile.getData(password)
          return true
        } catch {
          return false
        }
      }
    }
    ```
