  <name>Task 2: Create TagsSection component for side panel</name>
  <files>src/renderer/src/components/TagsSection.vue</files>
  <action>
    Create a component for displaying and managing tags in the side panel.

    ```vue
    <template>
      <div class="tags-section">
        <div class="d-flex align-center justify-space-between mb-2">
          <div class="text-subtitle-2">Tags</div>
          <v-btn
            v-if="!isAssigning"
            size="x-small"
            variant="text"
            prepend-icon="mdi-plus"
            :disabled="availableTags.length === 0"
            @click="startAssigning"
          >
            Add
          </v-btn>
        </div>

        <!-- Assigned tags as chips -->
        <div v-if="assignedTags.length > 0" class="d-flex flex-wrap ga-1 mb-2">
          <v-chip
            v-for="tag in assignedTags"
            :key="tag.id"
            :color="tag.color"
            size="small"
            closable
            @click:close="removeTag(tag)"
          >
            {{ tag.name }}
          </v-chip>
        </div>
        <div v-else-if="!isAssigning" class="text-caption text-medium-emphasis">
          No tags assigned
        </div>

        <!-- Tag assignment UI -->
        <v-expand-transition>
          <div v-if="isAssigning" class="mt-2">
            <v-select
              v-model="selectedTagIds"
              :items="unassignedTags"
              item-title="name"
              item-value="id"
              multiple
              chips
              density="compact"
              variant="outlined"
              hide-details
              placeholder="Select tags to add..."
              class="mb-2"
            >
              <template #chip="{ item, props: chipProps }">
                <v-chip
                  v-bind="chipProps"
                  :color="item.raw.color"
                  size="x-small"
                >
                  {{ item.raw.name }}
                </v-chip>
              </template>
              <template #item="{ item, props: itemProps }">
                <v-list-item v-bind="itemProps">
                  <template #prepend>
                    <div
                      class="tag-color-dot mr-2"
                      :style="{ backgroundColor: item.raw.color }"
                    />
                  </template>
                </v-list-item>
              </template>
            </v-select>

            <div class="d-flex ga-2">
              <v-btn
                size="small"
                color="primary"
                variant="flat"
                :disabled="selectedTagIds.length === 0"
                :loading="loading"
                @click="applyTags"
              >
                Apply
              </v-btn>
              <v-btn size="small" variant="outlined" @click="cancelAssigning">
                Cancel
              </v-btn>
            </div>
          </div>
        </v-expand-transition>

        <!-- No tags exist hint -->
        <div v-if="availableTags.length === 0" class="text-caption text-medium-emphasis mt-2">
          <v-icon size="x-small" class="mr-1">mdi-information-outline</v-icon>
          Create tags in Settings > Custom Tags
        </div>
      </div>
    </template>

    <script setup lang="ts">
    import { ref, computed, watch, onMounted } from 'vue'
    import { useTags, type Tag } from '../composables/useTags'

    const props = defineProps<{
      caseId: number
      variantId: number
    }>()

    const emit = defineEmits<{
      'tags-changed': []
    }>()

    const { tags: availableTags, loadTags, getTagsForVariant, assignTag, unassignTag } = useTags()

    const assignedTags = ref<Tag[]>([])
    const isAssigning = ref(false)
    const selectedTagIds = ref<number[]>([])
    const loading = ref(false)

    // Tags not yet assigned to this variant
    const unassignedTags = computed(() => {
      const assignedIds = new Set(assignedTags.value.map(t => t.id))
      return availableTags.value.filter(t => !assignedIds.has(t.id))
    })

    // Load assigned tags for current variant
    const loadAssignedTags = async () => {
      assignedTags.value = await getTagsForVariant(props.caseId, props.variantId)
    }

    // Watch for variant changes
    watch(
      () => [props.caseId, props.variantId],
      async () => {
        await loadAssignedTags()
        isAssigning.value = false
        selectedTagIds.value = []
      },
      { immediate: true }
    )

    onMounted(async () => {
      await loadTags()
    })

    const startAssigning = () => {
      isAssigning.value = true
      selectedTagIds.value = []
    }

    const cancelAssigning = () => {
      isAssigning.value = false
      selectedTagIds.value = []
    }

    const applyTags = async () => {
      loading.value = true
      try {
        for (const tagId of selectedTagIds.value) {
          await assignTag(props.caseId, props.variantId, tagId)
        }
        await loadAssignedTags()
        emit('tags-changed')
      } finally {
        loading.value = false
        cancelAssigning()
      }
    }

    const removeTag = async (tag: Tag) => {
      loading.value = true
      try {
        await unassignTag(props.caseId, props.variantId, tag.id)
        await loadAssignedTags()
        emit('tags-changed')
      } finally {
        loading.value = false
      }
    }
    </script>

    <style scoped>
    .tags-section {
      padding: 12px 0;
    }

    .tag-color-dot {
      width: 12px;
      height: 12px;
      border-radius: 50%;
    }
    </style>
    ```
  </action>

  <name>Task 3: Create HpoAutocomplete component</name>
  <files>src/renderer/src/components/HpoAutocomplete.vue</files>
  <action>
    Create a reusable HPO term autocomplete using bundled JSON.

    ```vue
    <template>
      <v-autocomplete
        v-model="selectedTerms"
        :items="searchResults"
        :loading="loading"
        item-title="label"
        item-value="id"
        :label="label"
        :placeholder="placeholder"
        multiple
        chips
        closable-chips
        density="compact"
        variant="outlined"
        hide-details
        no-filter
        return-object
        @update:search="handleSearch"
      >
        <template #chip="{ item, props: chipProps }">
          <v-chip
            v-bind="chipProps"
            closable
            size="small"
            color="info"
          >
            {{ item.raw.label }}
          </v-chip>
        </template>
        <template #item="{ item, props: itemProps }">
          <v-list-item
            v-bind="itemProps"
            :title="item.raw.label.split(' - ')[1]"
            :subtitle="item.raw.id"
          />
        </template>
        <template #no-data>
          <v-list-item v-if="searchQuery.length < 2">
            <v-list-item-title class="text-caption text-medium-emphasis">
              Type at least 2 characters to search...
            </v-list-item-title>
          </v-list-item>
          <v-list-item v-else-if="searchQuery.length >= 2 && !loading">
            <v-list-item-title class="text-caption text-medium-emphasis">
              No matching HPO terms found
            </v-list-item-title>
          </v-list-item>
        </template>
      </v-autocomplete>
    </template>

    <script setup lang="ts">
    import { ref, watch } from 'vue'
    import { useHpoBundled } from '../composables/useHpoBundled'

    interface HpoSearchResult {
      id: string
      label: string
    }

    const props = withDefaults(defineProps<{
      modelValue?: HpoSearchResult[]
      label?: string
      placeholder?: string
    }>(), {
      modelValue: () => [],
      label: 'Search HPO terms',
      placeholder: 'Type to search (e.g., seizure, developmental delay...)'
    })

    const emit = defineEmits<{
      'update:modelValue': [value: HpoSearchResult[]]
    }>()

    const { loadHpoData, searchTerms, loading: hpoLoading } = useHpoBundled()

    const searchQuery = ref('')
    const searchResults = ref<HpoSearchResult[]>([])
    const loading = ref(false)

    // Sync with v-model
    const selectedTerms = ref<HpoSearchResult[]>([...props.modelValue])

    watch(
      () => props.modelValue,
      (newVal) => {
        selectedTerms.value = [...newVal]
      }
    )

    watch(selectedTerms, (newVal) => {
      emit('update:modelValue', newVal)
    })

    // Debounce timer
    let searchTimeout: ReturnType<typeof setTimeout> | null = null

    const handleSearch = async (query: string) => {
      searchQuery.value = query

      // Clear previous timeout
      if (searchTimeout) {
        clearTimeout(searchTimeout)
      }

      if (!query || query.length < 2) {
        searchResults.value = []
        return
      }

      // Debounce search by 200ms
      searchTimeout = setTimeout(async () => {
        loading.value = true
        try {
          // Lazy load HPO data on first search
          await loadHpoData()
          searchResults.value = searchTerms(query, 10)
        } finally {
          loading.value = false
        }
      }, 200)
    }
    </script>
    ```
  </action>
