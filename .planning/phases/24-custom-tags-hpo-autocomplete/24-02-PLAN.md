<task type="auto">
  <name>Task 2: Create TagManagementDialog component</name>
  <files>src/renderer/src/components/TagManagementDialog.vue</files>
  <action>
    Create the tag management dialog following ExternalLinksSettings.vue pattern.

    ```vue
    <template>
      <v-dialog v-model="isOpen" max-width="600" scrollable>
        <v-card>
          <v-card-title class="d-flex align-center justify-space-between">
            <span>Custom Tags</span>
            <v-btn
              color="primary"
              variant="flat"
              size="small"
              prepend-icon="mdi-plus"
              @click="startCreate"
            >
              New Tag
            </v-btn>
          </v-card-title>

          <v-card-text>
            <!-- Tag List -->
            <v-list v-if="tags.length > 0" density="compact" class="mb-4">
              <v-list-item
                v-for="tag in tags"
                :key="tag.id"
              >
                <template #prepend>
                  <v-chip :color="tag.color" size="small" class="mr-3">
                    {{ tag.name }}
                  </v-chip>
                </template>

                <v-list-item-title>{{ tag.name }}</v-list-item-title>

                <template #append>
                  <v-btn
                    icon="mdi-pencil"
                    size="x-small"
                    variant="text"
                    @click="startEdit(tag)"
                  />
                  <v-btn
                    icon="mdi-delete"
                    size="x-small"
                    variant="text"
                    color="error"
                    @click="confirmDelete(tag)"
                  />
                </template>
              </v-list-item>
            </v-list>

            <div v-else class="text-center text-medium-emphasis py-8">
              <v-icon size="48" class="mb-2">mdi-tag-off-outline</v-icon>
              <div>No tags created yet</div>
              <div class="text-caption">Click "New Tag" to create your first tag</div>
            </div>

            <!-- Create/Edit Form -->
            <v-expand-transition>
              <v-card v-if="editingTag !== null" variant="outlined" class="mt-4">
                <v-card-text>
                  <div class="text-subtitle-2 mb-3">
                    {{ isCreateMode ? 'Create New Tag' : `Edit "${editingTag.name}"` }}
                  </div>

                  <v-text-field
                    v-model="editForm.name"
                    label="Tag Name"
                    variant="outlined"
                    density="compact"
                    :error-messages="nameError"
                    class="mb-4"
                    autofocus
                  />

                  <div class="text-subtitle-2 mb-2">Color</div>
                  <ColorSwatchPicker v-model="editForm.color" />

                  <div class="d-flex ga-2 mt-4">
                    <v-btn
                      color="primary"
                      variant="flat"
                      :loading="loading"
                      :disabled="!isFormValid"
                      @click="saveTag"
                    >
                      {{ isCreateMode ? 'Create' : 'Save' }}
                    </v-btn>
                    <v-btn variant="outlined" @click="cancelEdit">Cancel</v-btn>
                  </div>
                </v-card-text>
              </v-card>
            </v-expand-transition>
          </v-card-text>

          <v-divider />

          <v-card-actions>
            <v-spacer />
            <v-btn color="primary" variant="flat" @click="isOpen = false">Close</v-btn>
          </v-card-actions>
        </v-card>

        <!-- Delete Confirmation Dialog -->
        <v-dialog v-model="deleteDialog" max-width="400">
          <v-card>
            <v-card-title>Delete Tag?</v-card-title>
            <v-card-text>
              <p>Are you sure you want to delete "{{ tagToDelete?.name }}"?</p>
              <p v-if="tagToDelete && deleteUsageCount > 0" class="text-warning mt-2">
                <v-icon size="small" class="mr-1">mdi-alert</v-icon>
                This tag is assigned to {{ deleteUsageCount }} variant{{ deleteUsageCount > 1 ? 's' : '' }}.
                Deleting will remove it from all variants.
              </p>
            </v-card-text>
            <v-card-actions>
              <v-spacer />
              <v-btn variant="text" @click="deleteDialog = false">Cancel</v-btn>
              <v-btn color="error" variant="flat" :loading="loading" @click="executeDelete">
                Delete
              </v-btn>
            </v-card-actions>
          </v-card>
        </v-dialog>
      </v-dialog>
    </template>

    <script setup lang="ts">
    import { ref, computed, onMounted } from 'vue'
    import ColorSwatchPicker from './ColorSwatchPicker.vue'
    import { useTags, type Tag } from '../composables/useTags'

    const { tags, loading, loadTags, createTag, updateTag, deleteTag, getTagUsageCount } = useTags()

    const isOpen = ref(false)

    // Edit state
    const editingTag = ref<Tag | { id: null } | null>(null)
    const isCreateMode = computed(() => editingTag.value !== null && editingTag.value.id === null)
    const editForm = ref({
      name: '',
      color: '#a09588' // Default to primary/taupe
    })

    // Delete state
    const deleteDialog = ref(false)
    const tagToDelete = ref<Tag | null>(null)
    const deleteUsageCount = ref(0)

    // Validation
    const nameError = computed(() => {
      if (editingTag.value !== null && editForm.value.name.trim() === '') {
        return 'Tag name is required'
      }
      // Check for duplicate names (excluding current tag being edited)
      const normalizedName = editForm.value.name.trim().toLowerCase()
      const duplicate = tags.value.find(
        t => t.name.toLowerCase() === normalizedName &&
             (editingTag.value === null || editingTag.value.id === null || t.id !== editingTag.value.id)
      )
      if (duplicate) {
        return 'A tag with this name already exists'
      }
      return ''
    })

    const isFormValid = computed(() => {
      return editForm.value.name.trim() !== '' && nameError.value === ''
    })

    // Actions
    const startCreate = () => {
      editingTag.value = { id: null }
      editForm.value = {
        name: '',
        color: '#a09588'
      }
    }

    const startEdit = (tag: Tag) => {
      editingTag.value = tag
      editForm.value = {
        name: tag.name,
        color: tag.color
      }
    }

    const cancelEdit = () => {
      editingTag.value = null
    }

    const saveTag = async () => {
      if (!isFormValid.value || editingTag.value === null) return

      const name = editForm.value.name.trim()
      const color = editForm.value.color

      if (isCreateMode.value) {
        await createTag(name, color)
      } else {
        await updateTag((editingTag.value as Tag).id, name, color)
      }

      cancelEdit()
    }

    const confirmDelete = async (tag: Tag) => {
      tagToDelete.value = tag
      deleteUsageCount.value = await getTagUsageCount(tag.id)
      deleteDialog.value = true
    }

    const executeDelete = async () => {
      if (tagToDelete.value === null) return

      await deleteTag(tagToDelete.value.id)
      deleteDialog.value = false
      tagToDelete.value = null
      deleteUsageCount.value = 0
    }

    // Load tags when dialog opens
    const show = async () => {
      isOpen.value = true
      cancelEdit()
      await loadTags(true) // Force refresh
    }

    onMounted(() => {
      // Pre-load tags
      loadTags()
    })

    defineExpose({ show })
    </script>
    ```
