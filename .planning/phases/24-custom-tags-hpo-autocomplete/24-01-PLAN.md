    ```typescript
    import { ipcMain } from 'electron'
    import type Database from 'better-sqlite3-multiple-ciphers'

    export interface Tag {
      id: number
      name: string
      color: string
      created_at: number
    }

    export interface VariantTag {
      id: number
      case_id: number
      variant_id: number
      tag_id: number
      created_at: number
    }

    export function registerTagsHandlers(db: Database.Database): void {
      // List all tags
      ipcMain.handle('tags:list', async () => {
        const stmt = db.prepare('SELECT * FROM tags ORDER BY name')
        return stmt.all() as Tag[]
      })

      // Create a new tag
      ipcMain.handle('tags:create', async (_, name: string, color: string) => {
        const stmt = db.prepare(`
          INSERT INTO tags (name, color, created_at)
          VALUES (?, ?, ?)
        `)
        const result = stmt.run(name, color, Date.now())
        return { id: result.lastInsertRowid as number, name, color, created_at: Date.now() }
      })

      // Update a tag
      ipcMain.handle('tags:update', async (_, tagId: number, name: string, color: string) => {
        const stmt = db.prepare(`
          UPDATE tags SET name = ?, color = ? WHERE id = ?
        `)
        stmt.run(name, color, tagId)
      })

      // Delete a tag (cascade deletes variant_tags via FK)
      ipcMain.handle('tags:delete', async (_, tagId: number) => {
        const stmt = db.prepare('DELETE FROM tags WHERE id = ?')
        stmt.run(tagId)
      })

      // Get usage count for a tag (for delete confirmation)
      ipcMain.handle('tags:count-usage', async (_, tagId: number) => {
        const stmt = db.prepare('SELECT COUNT(*) as count FROM variant_tags WHERE tag_id = ?')
        const result = stmt.get(tagId) as { count: number }
        return result.count
      })

      // Get tags assigned to a specific variant in a case
      ipcMain.handle('tags:get-for-variant', async (_, caseId: number, variantId: number) => {
        const stmt = db.prepare(`
          SELECT t.* FROM tags t
          INNER JOIN variant_tags vt ON t.id = vt.tag_id
          WHERE vt.case_id = ? AND vt.variant_id = ?
          ORDER BY t.name
        `)
        return stmt.all(caseId, variantId) as Tag[]
      })

      // Assign a tag to a variant (per-case)
      ipcMain.handle('tags:assign', async (_, caseId: number, variantId: number, tagId: number) => {
        const stmt = db.prepare(`
          INSERT OR IGNORE INTO variant_tags (case_id, variant_id, tag_id, created_at)
          VALUES (?, ?, ?, ?)
        `)
        stmt.run(caseId, variantId, tagId, Date.now())
      })

      // Remove a tag from a variant (per-case)
      ipcMain.handle('tags:unassign', async (_, caseId: number, variantId: number, tagId: number) => {
        const stmt = db.prepare(`
          DELETE FROM variant_tags
          WHERE case_id = ? AND variant_id = ? AND tag_id = ?
        `)
        stmt.run(caseId, variantId, tagId)
      })

      // Get tag counts for all variants in a case (for table display)
      ipcMain.handle('tags:get-counts-for-case', async (_, caseId: number) => {
        const stmt = db.prepare(`
          SELECT variant_id, COUNT(*) as tag_count
          FROM variant_tags
          WHERE case_id = ?
          GROUP BY variant_id
        `)
        return stmt.all(caseId) as Array<{ variant_id: number; tag_count: number }>
      })
    }
    ```

    ```typescript
    import { ref, readonly } from 'vue'

    export interface Tag {
      id: number
      name: string
      color: string
      created_at: number
    }

    // Module-level cache for tags list
    const tagsCache = ref<Tag[]>([])
    const tagsLoaded = ref(false)

    export function useTags() {
      const loading = ref(false)
      const error = ref<string | null>(null)

      // Load all tags (with cache)
      async function loadTags(forceRefresh = false): Promise<Tag[]> {
        if (tagsLoaded.value && !forceRefresh) {
          return tagsCache.value
        }

        loading.value = true
        error.value = null

        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          const tags = await (window as any).api.tags.list()
          tagsCache.value = tags
          tagsLoaded.value = true
          return tags
        } catch (e) {
          error.value = e instanceof Error ? e.message : 'Failed to load tags'
          return []
        } finally {
          loading.value = false
        }
      }

      // Create a new tag
      async function createTag(name: string, color: string): Promise<Tag | null> {
        loading.value = true
        error.value = null

        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          const tag = await (window as any).api.tags.create(name, color)
          // Refresh cache
          await loadTags(true)
          return tag
        } catch (e) {
          error.value = e instanceof Error ? e.message : 'Failed to create tag'
          return null
        } finally {
          loading.value = false
        }
      }

      // Update a tag
      async function updateTag(tagId: number, name: string, color: string): Promise<boolean> {
        loading.value = true
        error.value = null

        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          await (window as any).api.tags.update(tagId, name, color)
          // Refresh cache
          await loadTags(true)
          return true
        } catch (e) {
          error.value = e instanceof Error ? e.message : 'Failed to update tag'
          return false
        } finally {
          loading.value = false
        }
      }

      // Delete a tag
      async function deleteTag(tagId: number): Promise<boolean> {
        loading.value = true
        error.value = null

        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          await (window as any).api.tags.delete(tagId)
          // Refresh cache
          await loadTags(true)
          return true
        } catch (e) {
          error.value = e instanceof Error ? e.message : 'Failed to delete tag'
          return false
        } finally {
          loading.value = false
        }
      }

      // Get usage count for delete confirmation
      async function getTagUsageCount(tagId: number): Promise<number> {
        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          return await (window as any).api.tags.countUsage(tagId)
        } catch {
          return 0
        }
      }

      // Get tags for a specific variant in a case
      async function getTagsForVariant(caseId: number, variantId: number): Promise<Tag[]> {
        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          return await (window as any).api.tags.getForVariant(caseId, variantId)
        } catch {
          return []
        }
      }

      // Assign tag to variant
      async function assignTag(caseId: number, variantId: number, tagId: number): Promise<boolean> {
        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          await (window as any).api.tags.assign(caseId, variantId, tagId)
          return true
        } catch {
          return false
        }
      }

      // Remove tag from variant
      async function unassignTag(caseId: number, variantId: number, tagId: number): Promise<boolean> {
        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          await (window as any).api.tags.unassign(caseId, variantId, tagId)
          return true
        } catch {
          return false
        }
      }

      // Get tag counts for all variants in a case
      async function getTagCountsForCase(caseId: number): Promise<Map<number, number>> {
        try {
          // eslint-disable-next-line @typescript-eslint/no-explicit-any
          const counts = await (window as any).api.tags.getCountsForCase(caseId) as Array<{ variant_id: number; tag_count: number }>
          return new Map(counts.map(c => [c.variant_id, c.tag_count]))
        } catch {
          return new Map()
        }
      }

      // Clear cache (call when database changes)
      function clearCache(): void {
        tagsCache.value = []
        tagsLoaded.value = false
      }

      return {
        // State
        tags: readonly(tagsCache),
        loading: readonly(loading),
        error: readonly(error),

        // Tag CRUD
        loadTags,
        createTag,
        updateTag,
        deleteTag,
        getTagUsageCount,

        // Variant-tag operations
        getTagsForVariant,
        assignTag,
        unassignTag,
        getTagCountsForCase,

        // Cache management
        clearCache
      }
    }
    ```

    ```typescript
    import { ref } from 'vue'

    export interface HpoTerm {
      id: string
      name: string
    }

    // Module-level cache for lazy-loaded HPO data
    let hpoTerms: HpoTerm[] | null = null
    const loading = ref(false)

    export function useHpoBundled() {
      /**
       * Lazy-load HPO data from bundled JSON.
       * Only loads once; subsequent calls return immediately.
       */
      async function loadHpoData(): Promise<void> {
        if (hpoTerms !== null) return // Already loaded

        loading.value = true
        try {
          // Dynamic import for lazy loading - Vite handles this as a separate chunk
          const module = await import('../assets/data/hpo-terms.json')
          hpoTerms = module.default as HpoTerm[]
        } catch (error) {
          console.error('Failed to load HPO data:', error)
          hpoTerms = []
        } finally {
          loading.value = false
        }
      }

      /**
       * Search HPO terms by name (case-insensitive).
       * Per user decision: search official names only, not synonyms.
       *
       * @param query - Search string (min 2 characters recommended)
       * @param limit - Maximum results to return (default 10)
       * @returns Array of matching terms with id and label
       */
      function searchTerms(query: string, limit = 10): Array<{ id: string; label: string }> {
        if (!hpoTerms || query.length < 2) return []

        const normalizedQuery = query.toLowerCase().trim()
        const results: Array<{ id: string; label: string }> = []

        for (const term of hpoTerms) {
          if (results.length >= limit) break

          // Search in term name only (not synonyms, per user decision)
          if (term.name.toLowerCase().includes(normalizedQuery)) {
            results.push({
              id: term.id,
              label: `${term.id} - ${term.name}`
            })
          }
        }

        return results
      }

      /**
       * Check if HPO data is loaded
       */
      function isLoaded(): boolean {
        return hpoTerms !== null && hpoTerms.length > 0
      }

      /**
       * Get total term count (for diagnostics)
       */
      function getTermCount(): number {
        return hpoTerms?.length ?? 0
      }

      return {
        loadHpoData,
        searchTerms,
        isLoaded,
        getTermCount,
        loading
      }
    }
    ```
